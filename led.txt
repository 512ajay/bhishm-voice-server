
#include <Arduino.h>
#include <MD_MAX72xx.h>
#include <SPI.h>

// Change the hardware type if your module differs.
#define HARDWARE_TYPE MD_MAX72XX::FC16_HW
#define MAX_DEVICES 2

#define DATA_PIN 16
#define CLK_PIN  18
#define CS_PIN   17

MD_MAX72XX mx = MD_MAX72XX(
  HARDWARE_TYPE, DATA_PIN, CLK_PIN, CS_PIN, MAX_DEVICES
);

// Each 8-bit pattern represents one eye.
// Adjust these patterns to suit your matrix orientation.

const uint8_t EYE_NEUTRAL[8] = {
  0b00111100,
  0b01111110,
  0b11100111,
  0b11000011,
  0b11000011,
  0b11100111,
  0b01111110,
  0b00111100
};

const uint8_t EYE_HAPPY[8] = {
  0b00000000,
  0b00100100,
  0b01100110,
  0b11000011,
  0b11000011,
  0b01100110,
  0b00111100,
  0b00000000
};

const uint8_t EYE_THINKING[8] = {
  0b00111100,
  0b01111110,
  0b11100111,
  0b11000011,
  0b11001111,
  0b11111110,
  0b01111100,
  0b00000000
};

const uint8_t EYE_SAD[8] = {
  0b00000000,
  0b01100110,
  0b11000011,
  0b11000011,
  0b11100111,
  0b01111110,
  0b00111100,
  0b00000000
};

const uint8_t EYE_ANGRY[8] = {
  0b11000011,
  0b11100111,
  0b01111110,
  0b00111100,
  0b00111100,
  0b01100110,
  0b11000011,
  0b00000000
};

void showEyes(const uint8_t left[8], const uint8_t right[8]) {
  mx.clear();

  for (uint8_t row = 0; row < 8; row++) {
    mx.setRow(1, row, left[row]);
    mx.setRow(0, row, right[row]);
  }

  mx.update();
}

void neutral() {
  showEyes(EYE_NEUTRAL, EYE_NEUTRAL);
}

void happy() {
  showEyes(EYE_HAPPY, EYE_HAPPY);
}

void thinking() {
  // Make the eyes look different for a thinking expression.
  showEyes(EYE_THINKING, EYE_NEUTRAL);
}

void sad() {
  showEyes(EYE_SAD, EYE_SAD);
}

void angry() {
  showEyes(EYE_ANGRY, EYE_ANGRY);
}

void setup() {
  Serial.begin(115200);

  mx.begin();
  mx.control(MD_MAX72XX::INTENSITY, 4);
  mx.clear();

  neutral();

  Serial.println("Robot Eyes Ready");
  Serial.println("1=Neutral 2=Happy 3=Thinking");
  Serial.println("4=Sad 5=Angry");
}

void loop() {
  if (Serial.available()) {
    char c = Serial.read();

    switch (c) {
      case '1': neutral();  break;
      case '2': happy();    break;
      case '3': thinking(); break;
      case '4': sad();      break;
      case '5': angry();    break;
    }
  }
}
